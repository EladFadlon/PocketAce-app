# החבר׳ה הקבועים — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the "החבר׳ה הקבועים" section fully functional — save regular players to localStorage, select who's playing tonight from the Lobby, auto-add them to GameSetup, and allow adding new regulars from the results screen.

**Architecture:** A new `src/utils/regulars.ts` utility owns all localStorage access for the regulars list. `Lobby.tsx` reads from it and manages selection state locally. `App.tsx` threads selected names into `GameSetup` via the existing `existingPlayers` prop. `SettlementResults.tsx` calls the utility directly to add new regulars post-game.

**Tech Stack:** React 19, TypeScript strict, localStorage, Framer Motion (existing), CSS custom properties

- [ ] **Step 1: Create the file**

```typescript
const KEY = 'pocketAce_regulars';

export function getRegulars(): string[] {
  try {
    return JSON.parse(localStorage.getItem(KEY) || '[]');
  } catch {
    return [];
  }
}

export function addRegular(name: string): void {
  const current = getRegulars();
  if (!current.includes(name)) {
    localStorage.setItem(KEY, JSON.stringify([...current, name]));
  }
}

export function removeRegular(name: string): void {
  localStorage.setItem(KEY, JSON.stringify(getRegulars().filter(n => n !== name)));
}
```

```typescript
const handleToggleSelect = (name: string) => {
  setPendingRemove(null);
  setSelectedRegulars(prev => {
    const next = new Set(prev);
    if (next.has(name)) next.delete(name);
    else next.add(name);
    return next;
  });
};

const handleLongPressStart = (name: string) => {
  longPressTimer.current = setTimeout(() => setPendingRemove(name), 500);
};

const handleLongPressEnd = () => {
  if (longPressTimer.current) clearTimeout(longPressTimer.current);
};

const handleRemoveRegular = (name: string) => {
  removeRegular(name);
  setRegulars(getRegulars());
  setPendingRemove(null);
  setSelectedRegulars(prev => { const next = new Set(prev); next.delete(name); return next; });
};

const handleAddRegular = () => {
  const trimmed = newRegularName.trim();
  if (!trimmed) return;
  addRegular(trimmed);
  setRegulars(getRegulars());
  setNewRegularName('');
  setShowAddSheet(false);
};
```

```typescript
const handleStartFromLobby = useCallback((selectedNames: string[]) => {
  if (!currentUser) return;
  const hostPlayer: Player = {
    id: `host_${currentUser.id}`,
    name: currentUser.name,
    buyIns: 1,
    totalInvested: settings.nisPerBuyIn,
    finalChips: settings.chipsPerBuyIn,
    isHost: true,
  };
  const regularPlayers: Player[] = selectedNames.map((name, i) => ({
    id: `regular_${i}_${Date.now()}`,
    name,
    buyIns: 1,
    totalInvested: settings.nisPerBuyIn,
    finalChips: settings.chipsPerBuyIn,
    isHost: false,
  }));
  setPlayers([hostPlayer, ...regularPlayers]);
  setSettlements([]);
  setDebts([]);
  setHostFee(0);
  setPhase(GamePhaseEnum.SETUP);
}, [currentUser, settings]);
```
